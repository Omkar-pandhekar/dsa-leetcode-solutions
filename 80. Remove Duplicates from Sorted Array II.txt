class Solution {
    public int removeDuplicates(int[] nums) {
        int left = 0, right = 0;
        int count;
        
        if(nums.length == 2){
            return 2;
        }
        
        while(right < nums.length){
            count = 1;
            while(right+1 < nums.length && nums[right] == nums[right+1]){
                count += 1;
                right += 1;
            }
            
            for(int i = 0; i < Math.min(2,count);i++){
                nums[left] = nums[right];
                left += 1;
            }
            
            right += 1;
        }
        
        return left;
    }
}

class Solution {
    public int removeDuplicates(int[] nums) {
       int k = 2;
       
       for(int i=2;i<nums.length;i++){
            if(nums[i] != nums[k-2]){
                nums[k] = nums[i];
                k++;
            }
       }
       
       return k;
    }
}
